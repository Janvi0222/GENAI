from transformers import AutoProcessor, MusicgenForConditionalGeneration
from IPython.display import Audio

p = AutoProcessor.from_pretrained("facebook/musicgen-small")

m = MusicgenForConditionalGeneration.from_pretrained(
    "facebook/musicgen-small"
)

text = input("Enter music description: ")

x = p(text=text, return_tensors="pt")

y = m.generate(**x, max_new_tokens=256)

Audio(y[0].numpy(), rate=32000)
