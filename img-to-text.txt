from transformers import pipeline
from PIL import Image

captioner = pipeline(
    "image-text-to-text",
    model="Salesforce/blip-image-captioning-base"
)

image = Image.open("/content/mountain.jpg")

result = captioner(
    images=image,
    prompt="Describe this image."
)
